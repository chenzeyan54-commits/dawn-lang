# Private benchmark of production cold, record, admit, replay and renewal APIs.
# Checks run outside each operation timer; no private replay implementation.
# Snapshot construction and retained-heap probes have separate units and modes.
# Driven by bench-replay.py; stdout is strict meta/round TSV.

# ---------------------------------------------------------------- subjects

fn join(parts: List[String]) -> String = {
  var out = ""
  for p in parts { out = out ++ p ++ "\n" }
  out
}

fn show_diags(ds: List[Diag]) -> String = {
  var out = ""
  for d in ds { out = out ++ d.msg ++ " | " }
  out
}

## One module per class. Support declarations are a fixed O(1) prefix; the
## per-body cost divides by the number of bodies the scheduler actually ran.
pub fn subject(kind: String, n: Int) -> String = {
  var parts: List[String] = []
  if kind == "scalar" {
    for i in 0..n { parts = parts ++ ["fn s${i}() -> Int = ${i}"] }
  }
  if kind == "arith" {
    for i in 0..n {
      parts = parts ++ ["fn p${i}(a: Int, b: Int) -> Int = {\n  let c = a + b * 2\n  let d = c - a + ${i}\n  let e = if d > 0 { d } else { 0 - d }\n  e + c * 3 - d\n}"]
    }
  }
  if kind == "calls" {
    parts = parts ++ ["fn helper(x: Int) -> Int = x + 1", "fn helper2(x: Int, y: Int) -> Int = x * y - 1"]
    for i in 0..n {
      parts = parts ++ ["fn c${i}(a: Int) -> Int = helper(a) + helper2(a, ${i}) + helper(a + ${i})"]
    }
  }
  if kind == "primitive_inferred" {
    parts = parts ++ ["fn seed() = 1"]
    for i in 0..n { parts = parts ++ ["fn ip${i}() = seed() + ${i}"] }
  }
  if kind == "generic" {
    parts = parts ++ ["trait Scale[T] { fn scale(x: T, k: Int) -> Int }",
      "type Coin = { value: Int }",
      "impl Scale[Coin] { fn scale(x: Coin, k: Int) -> Int = x.value * k }"]
    for i in 0..n {
      parts = parts ++ ["fn g${i}[T: Scale](x: T, y: T) -> Int = scale(x, ${i}) + scale(y, 2) + scale(x, 3)"]
    }
  }
  if kind == "inferred" {
    for i in 0..n {
      parts = parts ++ ["fn q${i}(a: Int, b: Int) = {\n  let add = (x: Int, y: Int) => x + y\n  let twice = (f: fn(Int, Int) -> Int, x: Int) => f(x, x)\n  let t = twice(add, a + ${i})\n  let u = twice((x, y) => x * y, b)\n  t + u\n}"]
    }
  }
  if kind == "method" {
    parts = parts ++ ["trait Shine[T] { fn shine(x: T) -> Int }"]
    for i in 0..n {
      parts = parts ++ ["type W${i} = { v: Int }", "impl Shine[W${i}] { fn shine(x: W${i}) -> Int = x.v * 2 + ${i} }"]
    }
  }
  if kind == "default" {
    for i in 0..n {
      parts = parts ++ ["trait Buff${i}[T] { fn raw${i}(x: T) -> Int\n  fn buff${i}(x: T) -> Int = raw${i}(x) + ${i} }"]
    }
  }
  if kind == "java" {
    parts = parts ++ ["use java \"java.lang.Math\""]
    for i in 0..n {
      parts = parts ++ ["fn j${i}(x: Int) -> Int !io = Math.abs(x - ${i}) + Math.max(x, ${i})"]
    }
  }
  join(parts)
}

fn empty_env() -> Map[String, ModExports] = map.empty()

fn base_cx() -> Cx !io = Cx { ..cx_new(Some("sample"), Some("sample.dawn")), jsig: jsig_real() }

fn scope_key() -> ModuleKey = identity.module_key("bench", "sample", "sample", "sample.dawn")

## One parse per revision: the snapshot keeps the tree the headers are checked
## from, beside the code-point index replay projects through.
fn snapshot_of(text: String) -> Snapshot = match source_snapshot.of(text) {
  Some(s) -> s, None -> panic("subject parse")
}

fn headers_of_snapshot(s: Snapshot) -> ModuleHeaders !io = {
  let h = checker.check_module_headers(base_cx(), source_snapshot.syntax(s), empty_env())
  if h.cx.diags != [] { panic("subject headers: ${show_diags(h.cx.diags)}") }
  h
}

fn headers_of(text: String) -> ModuleHeaders !io = headers_of_snapshot(snapshot_of(text))

fn body_count(b: ModuleBodies) -> Int =
  len(b.functions) + len(b.constants) + len(b.impl_methods) + len(b.trait_defaults) + len(b.tests)

# ---------------------------------------------------------------- reporting

fn meta(key: String, value: String) -> Unit !io = println("meta\t${key}\t${value}")

fn row(index: Int, metric: String, nanos: Int) -> Unit !io =
  println("round\t${index}\t${metric}\t${nanos}")

# ---------------------------------------------------------------- modes

fn edited(text: String) -> String = "# edited elsewhere in the file\n" ++ text

fn require(condition: Bool, label: String) -> Unit =
  if not condition { panic("benchmark invariant: " ++ label) }

# Compare actual scheduler outputs, not another implementation of checking.
fn equal_bodies(actual: ModuleBodies, expected: ModuleBodies) -> Bool =
  actual.functions == expected.functions && actual.constants == expected.constants &&
    actual.impl_methods == expected.impl_methods && actual.trait_defaults == expected.trait_defaults &&
    actual.tests == expected.tests && actual.cx.diags == expected.cx.diags &&
    actual.cx.syms == expected.cx.syms && actual.cx.decl_slots == expected.cx.decl_slots &&
    actual.cx.frame == expected.cx.frame && actual.cx.fns == expected.cx.fns &&
    actual.cx.alias_resolved == expected.cx.alias_resolved &&
    actual.cx.current_tparams == expected.cx.current_tparams &&
    actual.cx.current_eff_vars == expected.cx.current_eff_vars &&
    actual.cx.current_tparam_bounds == expected.cx.current_tparam_bounds &&
    actual.cx.function_reads == expected.cx.function_reads && actual.cx.body_writes == expected.cx.body_writes &&
    actual.cx.owner == expected.cx.owner && actual.cx.owner_lo == expected.cx.owner_lo &&
    actual.cx.owner_decl == expected.cx.owner_decl && actual.cx.in_test == expected.cx.in_test &&
    actual.cx.const_cutoff == expected.cx.const_cutoff && actual.cx.loop_jumps == expected.cx.loop_jumps &&
    actual.cx.take_cell == expected.cx.take_cell

# Fixture expectations, not admission rules. The original inference-heavy
# lambda class remains unsupported; primitive inference has its own workload.
fn expected_hits(kind: String, n: Int) -> Int =
  if kind == "scalar" || kind == "arith" { n }
  else { if kind == "calls" { n + 2 }
  else { if kind == "primitive_inferred" { n + 1 } else { 0 } } }

fn checked_cold(headers: ModuleHeaders) -> ModuleBodies !io = {
  let result = checker.check_module_body_products(headers)
  if result.cx.diags != [] { panic("cold fixture: ${show_diags(result.cx.diags)}") }
  result
}

fn verify_replay(result: scalar_replay.Replayed, cold: ModuleBodies, hits: Int) -> Unit = {
  require(equal_bodies(result.bodies, cold), "equal_bodies(result.bodies, cold)")
  require(result.reused == hits, "result.reused == hits")
  require(result.checked == body_count(cold) - hits, "result.checked == body_count(cold) - hits")
  require(result.checked == result.cold_unadmitted + result.cold_rejected, "result.checked == result.cold_unadmitted + result.cold_rejected")
  require(result.cold_rejected == 0, "result.cold_rejected == 0")
}

fn run_cold(kind: String, n: Int, rounds: Int) -> Unit !io = {
  let text = subject(kind, n)
  let headers = headers_of(text)
  let expected = checked_cold(headers)
  for r in 0..rounds {
    let t0 = System.nanoTime()
    let bodies = checker.check_module_body_products(headers)
    let t1 = System.nanoTime()
    require(equal_bodies(bodies, expected), "equal_bodies(bodies, expected)")
    row(r, "cold", t1 - t0)
  }
  meta("bodies", "${body_count(expected)}")
  meta("source_bytes", "${str.len(text)}")
}

fn run_record(kind: String, n: Int, rounds: Int, only_admit: Bool) -> Unit !io = {
  let snapshot = snapshot_of(subject(kind, n))
  let headers = headers_of_snapshot(snapshot)
  let scope = scope_key()
  let expected = checked_cold(headers)
  let initial = body_execution.record(scope, headers)
  require(equal_bodies(initial.bodies, expected) && initial.refused == 0, "equal_bodies(initial.bodies, expected) && initial.refused == 0")
  require(initial.checked == body_count(expected), "initial.checked == body_count(expected)")
  for r in 0..rounds {
    if only_admit {
      let t0 = System.nanoTime()
      let admitted = scalar_replay.admit(scope, headers, snapshot, initial)
      let t1 = System.nanoTime()
      require(scalar_replay.admitted_count(admitted) == expected_hits(kind, n), "scalar_replay.admitted_count(admitted) == expected_hits(kind, n)")
      verify_replay(scalar_replay.replay(admitted, headers, snapshot), expected, expected_hits(kind, n))
      row(r, "admit", t1 - t0)
    } else {
      let t0 = System.nanoTime()
      let saved = body_execution.record(scope, headers)
      let t1 = System.nanoTime()
      require(equal_bodies(saved.bodies, expected), "equal_bodies(saved.bodies, expected)")
      require(saved.checked == body_count(expected) && saved.refused == 0, "saved.checked == body_count(expected) && saved.refused == 0")
      let admitted = scalar_replay.admit(scope, headers, snapshot, saved)
      require(scalar_replay.admitted_count(admitted) == expected_hits(kind, n), "scalar_replay.admitted_count(admitted) == expected_hits(kind, n)")
      verify_replay(scalar_replay.replay(admitted, headers, snapshot), expected, expected_hits(kind, n))
      row(r, "record", t1 - t0)
    }
  }
  meta("bodies", "${body_count(expected)}")
  meta("admitted", "${expected_hits(kind, n)}")
  meta("capture_refused", "0")
}

# Prebuild both revision inputs once. Neither parsing nor header checking is
# part of the body-operation timer; renewal alternates actual cache generations.
fn run_replay(kind: String, n: Int, rounds: Int, renew: Bool) -> Unit !io = {
  let before = snapshot_of(subject(kind, n))
  let after = snapshot_of(edited(subject(kind, n)))
  let old = headers_of_snapshot(before)
  let next = headers_of_snapshot(after)
  let old_cold = checked_cold(old)
  let next_cold = checked_cold(next)
  let saved = body_execution.record(scope_key(), old)
  require(equal_bodies(saved.bodies, old_cold) && saved.refused == 0, "equal_bodies(saved.bodies, old_cold) && saved.refused == 0")
  var admitted = scalar_replay.admit(scope_key(), old, before, saved)
  let hits = expected_hits(kind, n)
  require(scalar_replay.admitted_count(admitted) == hits, "scalar_replay.admitted_count(admitted) == hits")
  for r in 0..rounds {
    let backwards = renew && r % 2 == 1
    let headers = if backwards { old } else { next }
    let snapshot = if backwards { before } else { after }
    let expected = if backwards { old_cold } else { next_cold }
    if renew {
      let t0 = System.nanoTime()
      let result = scalar_replay.replay_and_record(admitted, headers, snapshot)
      let t1 = System.nanoTime()
      verify_replay(result.replayed, expected, hits)
      require(result.visited == body_count(expected) && result.capture_refused == 0, "result.visited == body_count(expected) && result.capture_refused == 0")
      require(scalar_replay.admitted_count(result.admitted) == hits, "scalar_replay.admitted_count(result.admitted) == hits")
      admitted = result.admitted
      row(r, "renew", t1 - t0)
    } else {
      let t0 = System.nanoTime()
      let result = scalar_replay.replay(admitted, headers, snapshot)
      let t1 = System.nanoTime()
      verify_replay(result, expected, hits)
      row(r, "replay", t1 - t0)
    }
  }
  meta("bodies", "${body_count(next_cold)}")
  meta("reused", "${hits}")
  meta("checked", "${body_count(next_cold) - hits}")
  meta("cold_unadmitted", "${body_count(next_cold) - hits}")
  meta("cold_rejected", "0")
  if renew {
    meta("visited", "${body_count(next_cold)}")
    meta("capture_refused", "0")
    meta("renewed_admitted", "${scalar_replay.admitted_count(admitted)}")
  }
}

## Parse, historical standalone index construction, syntax comparison and the
## actual single-parse snapshot constructor are timed independently. None is
## included in replay's body timer; no synthetic multiple is a production phase.
fn run_parse(kind: String, n: Int, rounds: Int) -> Unit !io = {
  let text = subject(kind, n)
  let headers = headers_of(text)
  for r in 0..rounds {
    let t0 = System.nanoTime()
    let (syntax, ds) = parser.parse_module(text)
    let t1 = System.nanoTime()
    let indexed = source_projection.index(text)
    let t2 = System.nanoTime()
    let same = ds == [] && syntax == headers.syntax
    let t3 = System.nanoTime()
    if not same { panic("reparse did not match the header syntax") }
    let no_spans: List[(Int, Int, Int)] = []
    if source_projection.tokens_of(indexed, no_spans) == None {
      panic("index did not bind to its own snapshot")
    }
    let t4 = System.nanoTime()
    let whole = snapshot_of(text)
    let t5 = System.nanoTime()
    if source_snapshot.syntax(whole) != headers.syntax { panic("snapshot tree did not match the headers") }
    row(r, "parse", t1 - t0)
    row(r, "index", t2 - t1)
    row(r, "compare", t3 - t2)
    row(r, "snapshot_of", t5 - t4)
  }
  meta("source_bytes", "${str.len(text)}")
  meta("decls", "${len(headers.syntax.decls)}")
}

## Retained cost, attributed rather than estimated: settle the heap, read used
## bytes, retain one more value, settle and read again.
##
## These metrics retain fresh indexes and whole snapshots, respectively.
## A whole snapshot includes syntax, code points and token facts. Their
## difference is not a measured marginal semantic-cache cost: ownership and
## sharing differ from a production session. GC deltas may include noise.
fn used_heap() -> Int !io = {
  let rt = Runtime.getRuntime().expect("runtime")
  var i = 0
  while i < 4 {
    System.gc()
    i = i + 1
  }
  rt.totalMemory() - rt.freeMemory()
}

fn run_memory(kind: String, n: Int, rounds: Int) -> Unit !io = {
  let text = subject(kind, n)
  var indexes: List[source_projection.Indexed] = []
  for r in 0..rounds {
    let before = used_heap()
    indexes = indexes ++ [source_projection.index(text)]
    let after = used_heap()
    row(r, "retained_index", after - before)
  }
  var keep: List[Snapshot] = []
  for r in 0..rounds {
    let before = used_heap()
    keep = keep ++ [snapshot_of(text)]
    let after = used_heap()
    row(r, "retained_snapshot", after - before)
  }
  if len(keep) != rounds || len(indexes) != rounds { panic("values were not retained") }
  meta("source_bytes", "${str.len(text)}")
}

## Clock-pair probe only; do not subtract it from whole-operation medians.
fn run_timer(kind: String, n: Int, rounds: Int) -> Unit !io = {
  for r in 0..rounds {
    let t0 = System.nanoTime()
    var sink = 0
    for i in 0..n {
      let a = System.nanoTime()
      let b = System.nanoTime()
      sink = sink + (b - a)
    }
    let t1 = System.nanoTime()
    if sink < 0 { panic("timer sink") }
    row(r, "timer", t1 - t0)
  }
  meta("pairs", "${n}")
}

pub fn main() -> Unit !io = {
  let a = args()
  if len(a) != 4 { panic("usage: <cold|record|admit|replay|renew|parse|memory|timer> <class> <n> <rounds>") }
  let mode = a[0]
  let kind = a[1]
  let n = Integer.parseInt(a[2])
  let rounds = Integer.parseInt(a[3])
  require(n > 0 && rounds > 0, "n > 0 && rounds > 0")
  meta("mode", mode)
  meta("class", kind)
  meta("n", "${n}")
  meta("rounds", "${rounds}")
  meta("java", match System.getProperty("java.version") { Some(v) -> v, None -> "unknown" })
  match mode {
    "cold" -> run_cold(kind, n, rounds)
    "record" -> run_record(kind, n, rounds, false)
    "admit" -> run_record(kind, n, rounds, true)
    "replay" -> run_replay(kind, n, rounds, false)
    "renew" -> run_replay(kind, n, rounds, true)
    "parse" -> run_parse(kind, n, rounds)
    "memory" -> run_memory(kind, n, rounds)
    "timer" -> run_timer(kind, n, rounds)
    _ -> panic("unknown mode ${mode}")
  }
}
