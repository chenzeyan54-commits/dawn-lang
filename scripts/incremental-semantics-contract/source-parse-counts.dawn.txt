# Each exported sample calls the production API once. The host resets counters
# after class initialization; semantic oracles that parse again stay outside
# this fixture and its measured invocation interval.
use compiler/check/source_snapshot

pub fn clean_on() -> Bool = {
  let parsed = source_snapshot.parse("# 🌅\nfn value() -> Int = 42\n", true)
  parsed.diagnostics == [] && len(parsed.syntax.decls) == 1 &&
    match parsed.snapshot { Some(_) -> true, None -> false }
}

pub fn clean_off() -> Bool = {
  let parsed = source_snapshot.parse("# 🌅\nfn value() -> Int = 42\n", false)
  parsed.diagnostics == [] && len(parsed.syntax.decls) == 1 && parsed.snapshot == None
}

pub fn recovered() -> Bool = {
  let parsed = source_snapshot.parse("fn broken( -> Int = 42\nfn after() -> Int = 1\n", true)
  parsed.diagnostics != [] && len(parsed.syntax.decls) > 0 && parsed.snapshot == None
}

pub fn lexer_error() -> Bool = {
  let parsed = source_snapshot.parse("fn first() -> String = \"oops\n", true)
  parsed.diagnostics != [] && parsed.snapshot == None
}

pub fn delegated_of() -> Bool =
  match source_snapshot.of("fn value() -> Int = 42\n") { Some(_) -> true, None -> false }

pub fn main() -> Unit = ()
